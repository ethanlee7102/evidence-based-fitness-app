# Lean dependency set for CI (lint + offline unit tests only).
#
# Deliberately self-contained — does NOT `-r requirements.txt`, which would pull
# the full ingestion/ML stack (docling, torch, supabase, fastapi, ...) and make
# every CI run multi-minute. The offline judge unit tests import only
# src/core/eval/judge.py -> anthropic_provider / llm_provider -> config, whose
# import chain needs nothing heavier than httpx + python-dotenv + stdlib.
#
# The LLM eval suite (pytest -m eval) is intentionally NOT run in CI: it costs
# money, needs API keys, and is non-deterministic. CI runs `pytest -m "not eval"`.
#
# Versions pinned to match requirements-dev.txt.
ruff==0.4.4
pytest==8.2.0
pytest-asyncio==0.23.7
httpx
python-dotenv
pydantic  # imported via src/schema/rag.py in the eval package import chain
